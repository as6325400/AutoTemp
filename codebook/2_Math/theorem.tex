\begin{itemize}
\item Cramer 法則
$$
\begin{aligned}ax+by=e\\cx+dy=f\end{aligned}
\Rightarrow
\begin{aligned}x=\dfrac{ed-bf}{ad-bc}\\y=\dfrac{af-ec}{ad-bc}\end{aligned}
$$

\item Vandermonde 恆等式
$$
C(n+m,k)=\sum_{i=0}^{k} C(n,i)\,C(m,k-i)
$$

\item 隔板法＋排容原理（容斥原理）

設整數 $n\ge0$、$k\ge1$，將 $n$ 個相同物品分給 $k$ 個不同對象。
約定 $\binom ab=0$ 若 $a<0$、$b<0$ 或 $b>a$。
\[
\begin{aligned}
x_1+\cdots+x_k=n,\ x_i\ge0 &: \binom{n+k-1}{k-1},\\
x_1+\cdots+x_k=n,\ x_i\ge1 &: \binom{n-1}{k-1}.
\end{aligned}
\]
若有整數下界 $x_i\ge a_i$，先令 $y_i=x_i-a_i$，總和改為 $n-\sum_i a_i$。
排容原理：扣掉不合法情況，再補回重複扣除的交集。
\[
\left|\bigcup_{i=1}^{k}A_i\right|
=\sum_{\emptyset\ne S\subseteq\{1,\ldots,k\}}
(-1)^{|S|+1}\left|\bigcap_{i\in S}A_i\right|.
\]
若限制 $0\le x_i\le b_i$（$b_i$ 為非負整數），解數為
\[
\sum_{S\subseteq\{1,\ldots,k\}}(-1)^{|S|}
\binom{n-\sum_{i\in S}(b_i+1)+k-1}{k-1}.
\]
$S$ 表示指定違反上界的變數；枚舉 $2^k$ 個子集合，適合 $k$ 小時使用。

\item Lucas 定理

設 $p$ 為質數、$n,k$ 為非負整數，將 $n,k$ 寫成 $p$ 進位：
$n=\sum_i n_i p^i$、$k=\sum_i k_i p^i$，則
\[
\binom nk\equiv\prod_i\binom{n_i}{k_i}\pmod p.
\]
實作時反覆取 $n\bmod p$、$k\bmod p$，再各自除以 $p$；
任一位 $k_i>n_i$ 時答案為 $0$。
$p$ 較小時，可用 $O(p)$ 預處理 $0$ 到 $p-1$ 的階乘與反階乘，
每次查詢只需逐位相乘；不直接適用於合數模數。

\item Legendre 公式

設 $p$ 為質數、$n\ge0$，$n!$ 中質因數 $p$ 的次方為
\[
\nu_p(n!)=\sum_{j\ge1}\left\lfloor\frac{n}{p^j}\right\rfloor.
\]
實作時令 $t=n$，反覆令 $t=\lfloor t/p\rfloor$ 並累加，直到 $t=0$；
避免直接計算 $p^j$ 溢位。
組合數的次方為 $\nu_p\!\binom nk=\nu_p(n!)-\nu_p(k!)-\nu_p((n-k)!)$（$0\le k\le n$）；
$n!$ 的十進位尾零數為 $\nu_5(n!)$。

\item 錯排公式

$D_n$ 為 $n$ 個不同元素的排列中，沒有任何元素留在原位的方案數。
$D_0=1$、$D_1=0$，且
\[
D_n=(n-1)(D_{n-1}+D_{n-2})\quad(n\ge2).
\]
用前兩項遞推，時間 $O(n)$、額外空間 $O(1)$。
可逐步取模；加法與乘法仍須防溢位。

\item Kirchhoff 定理

設 $L$ 為圖 $G$ 的 $n\times n$ Laplacian 矩陣，其中 $L_{ii}=d(i)$，$L_{ij}=-c$，$c$ 為邊 $(i,j)$ 的條數。
\begin{itemize}
    \item 無向圖的生成樹數為 $\lvert \det(\tilde{L}_{11}) \rvert$。
    \item 以 $r$ 為根的有向生成樹數為 $\lvert \det(\tilde{L}_{rr}) \rvert$。
\end{itemize}

\item Tutte 矩陣

令 $D$ 為 $n\times n$ 矩陣，當 $i<j$ 且 $(i,j)\in E$ 時令 $d_{ij}=x_{ij}$（$x_{ij}$ 由均勻隨機選取），否則令 $d_{ij}=-d_{ji}$。則 $\dfrac{\mathrm{rank}(D)}{2}$ 為圖 $G$ 的最大匹配數。

\item Cayley 公式

\begin{itemize}
  \item 對於帶標號頂點、度數序列為 $d_1,d_2,\ldots,d_n$ 的情形，生成樹數為 $\dfrac{(n-2)!}{(d_1-1)!(d_2-1)!\cdots(d_n-1)!}$。
  \item 設 $T_{n,k}$ 為在 $n$ 頂點、$k$ 個連通分量的帶標號森林數，且頂點 $1,2,\ldots,k$ 分屬不同分量，則 $T_{n,k}=k\,n^{\,n-k-1}$。
\end{itemize}

\item Erdős–Gallai 定理

非負整數序列 $d_1\ge\cdots\ge d_n$ 可表示為 $n$ 頂點簡單圖的度數序列，當且僅當 $d_1+\cdots+d_n$ 為偶數，且對每個 $1\le k\le n$ 皆有
\[
\textstyle\sum_{i=1}^k d_i \le k(k-1)+\sum_{i=k+1}^n \min(d_i,k)\,.
\]

\item Gale–Ryser 定理

非負整數序列 $a_1\ge\cdots\ge a_n$ 與 $b_1,\ldots,b_n$ 構成雙圖度數序列（bigraphic）的充要條件為
\[
\textstyle\sum_{i=1}^n a_i=\sum_{i=1}^n b_i
\quad\text{且}\quad
\sum_{i=1}^k a_i\le \sum_{i=1}^n\min(b_i,k)\ \text{對所有 }1\le k\le n.
\]

\item Fulkerson–Chen–Anstee 定理

一序列 $(a_1,b_1),\ldots,(a_n,b_n)$（非負整數對，且 $a_1\ge\cdots\ge a_n$）為有向圖可實現之入出度序列，當且僅當
\[
\begin{aligned}
\sum_{i=1}^n a_i &= \sum_{i=1}^n b_i,\\
\sum_{i=1}^k a_i &\le \sum_{i=1}^k\min(b_i,k-1)
  +\sum_{i=k+1}^n\min(b_i,k).
\end{aligned}
\]
其中不等式須對所有 $1\le k\le n$ 成立。

\item 測量師公式（鞋帶公式）

設單純多邊形的頂點沿邊界依序為
$P_i=(x_i,y_i)$，$P_{n+1}=P_1$，則
\[
S=\sum_{i=1}^{n}(x_i y_{i+1}-x_{i+1}y_i),
\qquad A=\frac{|S|}{2}.
\]
可凹但不可自交；$S>0$ 為逆時針，$S<0$ 為順時針。
整數座標可保留 $|S|=2A$；叉積與累加須防溢位。

\item 邊界格點數

沿用上述頂點順序，整數頂點的單純多邊形之邊界格點數為
\[
B=\sum_{i=1}^{n}\gcd\!\left(|x_{i+1}-x_i|,\ |y_{i+1}-y_i|\right).
\]
單一線段含兩端點有 $\gcd(|\Delta x|,|\Delta y|)+1$ 個格點；
多邊形各邊只加 $\gcd$，避免重複計算頂點。
搭配測量師公式與 Pick 定理可求內部格點數。

\item Pick 定理

對所有頂點為整數點的單純多邊形，有
\[
A=\#\{\text{內部格點}\}+\frac{\#\{\text{邊界格點}\}}{2}-1.
\]

\item 莫比烏斯反演公式

\begin{itemize}
  \item $f(n)=\sum_{d\mid n}g(d)\ \Leftrightarrow\ g(n)=\sum_{d\mid n}\mu(d)\,f\!\left(\dfrac{n}{d}\right)$
  \item $f(n)=\sum_{n\mid d}g(d)\ \Leftrightarrow\ g(n)=\sum_{n\mid d}\mu\!\left(\dfrac{d}{n}\right)f(d)$
\end{itemize}

\item 整除分塊

設整數 $n\ge1$，從 $l=1$ 開始，每次計算
\[
q=\left\lfloor\frac nl\right\rfloor,\qquad
r=\left\lfloor\frac nq\right\rfloor.
\]
則 $i\in[l,r]$ 的 $\lfloor n/i\rfloor$ 都等於 $q$，且 $r$ 為最右端。
若 $r=n$ 即結束，否則令 $l=r+1$ 繼續，共 $O(\sqrt n)$ 段。
若 $A(t)=\sum_{i=1}^{t}a_i$ 可快速求得，則
$\sum_{i=l}^{r}a_i g(\lfloor n/i\rfloor)=[A(r)-A(l-1)]g(q)$；
適合計算帶整除商的和式，乘法與累加須防溢位。

\item 球冠（Spherical cap）

\begin{itemize}
  \item 球面被平面切下的一部分。
  \item $r$: 球半徑，$a$: 球冠底面半徑，$h$: 球冠高度，$\theta=\arcsin(a/r)$。
  \item 體積 $=\dfrac{\pi h^2(3r-h)}{3}=\dfrac{\pi h(3a^2+h^2)}{6}=\dfrac{\pi r^3(2+\cos\theta)(1-\cos\theta)^2}{3}$。
  \item 表面積 $=2\pi rh=\pi(a^2+h^2)=2\pi r^2(1-\cos\theta)$。
\end{itemize}

\item 拉格朗日乘數法

\begin{itemize}
  \item 在 $k$ 個約束 $g_i(x_1,\ldots,x_n)=0$ 下優化 $f(x_1,\ldots,x_n)$。
  \item 拉格朗日函數 $\mathcal{L}(x_1,\ldots,x_n,\lambda_1,\ldots,\lambda_k)=f(x_1,\ldots,x_n)-\sum_{i=1}^{k}\lambda_i\,g_i(x_1,\ldots,x_n)$。
  \item 對應原始受限問題的解是拉格朗日函數的一個鞍點。
\end{itemize}
